/**

* ユーザーのGoogle Driveからフォルダ一覧を取得

* @return {Array} フォルダ情報の配列

*/

function getUserFolders() {

try {

const folders = [];

const allFolders = DriveApp.getFolders();

let count = 0;

const maxFolders = 100; // 最大100個のフォルダを取得

while (allFolders.hasNext() && count < maxFolders) {

try {

const folder = retryDriveOperation(function() {

return allFolders.next();

}, 3, 500);

folders.push({

id: folder.getId(),

name: folder.getName()

});

count++;

// レート制限を避けるため、少し待機

if (count % 10 === 0) {

Utilities.sleep(200);

}

} catch (error) {

Logger.log('フォルダ取得エラー（スキップ）: ' + error);

count++;

continue;

}

}

// 名前順にソート

folders.sort(function(a, b) {

return a.name.localeCompare(b.name, 'ja');

});

Logger.log('フォルダ取得完了: ' + folders.length + '個');

return folders;

} catch (error) {

Logger.log('getUserFolders エラー: ' + error);

return [];

}

}

/**

* Google Drive API呼び出しをリトライ付きで実行

* @param {Function} operation - 実行する操作

* @param {number} maxRetries - 最大リトライ回数（デフォルト: 5）

* @param {number} baseDelay - ベース遅延時間（ミリ秒、デフォルト: 1000）

* @return {*} 操作の結果

*/

function retryDriveOperation(operation, maxRetries, baseDelay) {

maxRetries = maxRetries || 5;

baseDelay = baseDelay || 1000;

for (let attempt = 0; attempt < maxRetries; attempt++) {

try {

return operation();

} catch (error) {

const errorMessage = error.toString();

// レート制限エラーかチェック

const isRateLimitError = errorMessage.includes('混雑') ||

errorMessage.includes('集中') ||

errorMessage.includes('rate limit') ||

errorMessage.includes('quota') ||

errorMessage.includes('429') ||

errorMessage.includes('503');

if (isRateLimitError && attempt < maxRetries - 1) {

// 指数バックオフで待機（2^attempt秒）

const delay = baseDelay * Math.pow(2, attempt);

Logger.log('Drive API レート制限エラー。' + delay + 'ms待機してリトライします（試行 ' + (attempt + 1) + '/' + maxRetries + '）');

Utilities.sleep(delay);

continue;

}

// レート制限エラーでない、または最後の試行の場合はエラーをスロー

throw error;

}

}

throw new Error('最大リトライ回数に達しました');

}

  

/**

* Google Driveフォルダから音声ファイルを取得

* @param {string} folderId - Google DriveフォルダID

* @return {Object} 音声ファイルのリスト

*/

function getAudioFiles(folderId) {

try {

Logger.log('getAudioFiles 開始: folderId = ' + folderId);

// フォルダIDの検証

if (!folderId || folderId.trim() === '') {

return {

success: false,

error: 'フォルダIDが指定されていません'

};

}

// リトライ付きでフォルダを取得

const folder = retryDriveOperation(function() {

return DriveApp.getFolderById(folderId);

});

if (!folder) {

Logger.log('フォルダが見つかりませんでした: ' + folderId);

return {

success: false,

error: 'フォルダが見つかりませんでした。フォルダIDが正しいか、アクセス権限があるか確認してください。'

};

}

Logger.log('フォルダ取得成功: ' + folder.getName());

  

// 音声ファイルのMIMEタイプ（拡張子も含めて判定）

const audioMimeTypes = [

'audio/mpeg',

'audio/mp3',

'audio/wav',

'audio/wave',

'audio/x-wav',

'audio/mp4', // .m4aファイルは多くの場合このMIMEタイプになる

'audio/m4a',

'audio/x-m4a',

'audio/ogg',

'audio/webm',

'audio/flac',

'audio/aac',

'audio/amr',

'audio/x-ms-wma',

'application/octet-stream', // 一部の音声ファイルはこのMIMEタイプになることがある

'video/mp4', // 一部の.m4aファイルはvideo/mp4として認識されることがある

'audio/quicktime' // QuickTime形式の音声ファイル

];

// 音声ファイルの拡張子

const audioExtensions = [

'.mp3', '.wav', '.m4a', '.m4p', '.aac', '.ogg', '.oga', '.webm',

'.flac', '.amr', '.wma', '.mp4', '.3gp', '.aiff', '.aif'

];

  

// リトライ付きでファイルを取得

const files = retryDriveOperation(function() {

return folder.getFiles();

});

const audioFiles = [];

const timeZone = 'Asia/Tokyo';

let fileCount = 0;

const maxFiles = 100; // 一度に処理する最大ファイル数

let allFilesCount = 0; // 全ファイル数（デバッグ用）

  

while (files.hasNext() && fileCount < maxFiles) {

try {

allFilesCount++;

// 各ファイル取得時にリトライ

const file = retryDriveOperation(function() {

return files.next();

}, 3, 500); // ファイル取得は短い待機時間でリトライ

const fileName = file.getName();

const mimeType = retryDriveOperation(function() {

return file.getMimeType();

}, 3, 500);

Logger.log('ファイル確認: ' + fileName + ' (MIME: ' + mimeType + ')');

// MIMEタイプで判定

const isAudioByMime = audioMimeTypes.indexOf(mimeType) !== -1;

// 拡張子で判定（MIMEタイプが正しくない場合のフォールバック）

const fileExtension = fileName.substring(fileName.lastIndexOf('.')).toLowerCase();

const isAudioByExtension = audioExtensions.indexOf(fileExtension) !== -1;

// ファイル名に音声ファイルのキーワードが含まれているかチェック（追加の判定）

const hasAudioKeyword = /audio|sound|音声|オーディオ|ミュージック|music/i.test(fileName);

// デバッグ情報を出力

Logger.log('判定結果: ' + fileName + ' - MIME判定: ' + isAudioByMime + ', 拡張子判定: ' + isAudioByExtension + ' (拡張子: ' + fileExtension + ', MIME: ' + mimeType + ')');

// 拡張子が音声ファイルの拡張子である場合、MIMEタイプに関係なく音声ファイルとして扱う

if (isAudioByMime || isAudioByExtension) {

Logger.log('音声ファイルとして認識: ' + fileName);

const modifiedDate = Utilities.formatDate(

retryDriveOperation(function() {

return file.getLastUpdated();

}, 3, 500),

timeZone,

'yyyy-MM-dd HH:mm:ss'

);

  

audioFiles.push({

id: file.getId(),

name: fileName,

size: file.getSize(),

mimeType: mimeType,

modifiedDate: modifiedDate,

status: 'waiting'

});

} else {

Logger.log('音声ファイルではない（スキップ）: ' + fileName + ' (MIME: ' + mimeType + ', 拡張子: ' + fileExtension + ')');

}

fileCount++;

// ファイル間で少し待機してレート制限を回避

if (fileCount % 10 === 0) {

Utilities.sleep(200);

}

} catch (error) {

Logger.log('ファイル取得エラー（スキップ）: ' + error);

// エラーが発生したファイルはスキップして続行

fileCount++;

continue;

}

}

Logger.log('全ファイル数: ' + allFilesCount + ', 音声ファイル数: ' + audioFiles.length);

  

// 更新日時順にソート（新しい順）

audioFiles.sort(function(a, b) {

return new Date(b.modifiedDate) - new Date(a.modifiedDate);

});

  

Logger.log('音声ファイル取得完了: ' + audioFiles.length + '個');

// 音声ファイルが見つからなかった場合の詳細情報

if (audioFiles.length === 0) {

// フォルダ内の全ファイル名を取得（デバッグ用）

const allFileNames = [];

try {

const allFiles = retryDriveOperation(function() {

return folder.getFiles();

});

let count = 0;

while (allFiles.hasNext() && count < 20) {

try {

const f = retryDriveOperation(function() {

return allFiles.next();

}, 2, 300);

allFileNames.push(f.getName() + ' (' + f.getMimeType() + ')');

count++;

} catch (e) {

break;

}

}

} catch (e) {

Logger.log('全ファイル名取得エラー: ' + e);

}

return {

success: true,

files: [],

error: '音声ファイルが見つかりませんでした。フォルダ内に音声ファイル（.mp3, .wav, .m4a等）が存在するか確認してください。全' + allFilesCount + '個のファイルを確認しました。',

totalFilesChecked: allFilesCount,

sampleFiles: allFileNames.slice(0, 5) // 最初の5ファイルをサンプルとして返す

};

}

return {

success: true,

files: audioFiles,

message: fileCount >= maxFiles ? 'ファイル数が多いため、最初の' + maxFiles + '件のみ表示しています' : null

};

} catch (error) {

Logger.log('getAudioFiles エラー: ' + error);

Logger.log('エラースタック: ' + error.stack);

const errorMessage = error.toString();

// レート制限エラーの場合、より分かりやすいメッセージを返す

if (errorMessage.includes('混雑') || errorMessage.includes('集中') || errorMessage.includes('rate limit')) {

return {

success: false,

error: 'Google Driveへのアクセスが集中しています。しばらく待ってから再度お試しください。'

};

}

// 権限エラーの場合

if (errorMessage.includes('permission') || errorMessage.includes('権限') || errorMessage.includes('access')) {

return {

success: false,

error: 'フォルダへのアクセス権限がありません。フォルダの共有設定を確認してください。'

};

}

// フォルダが見つからない場合

if (errorMessage.includes('not found') || errorMessage.includes('見つかりません')) {

return {

success: false,

error: 'フォルダが見つかりませんでした。フォルダIDが正しいか確認してください。'

};

}

return {

success: false,

error: 'ファイルの読み込みに失敗しました: ' + errorMessage

};

}

}

  

/**

* 音声ファイルを分割して文字起こし

* @param {string} fileId - Google DriveファイルID

* @param {number} splitMinutes - 分割時間（分）

* @param {string} prompt - カスタムプロンプト

* @param {string} mode - 文字起こしモード（'fast' または 'normal'）

* @param {boolean} infiniteTranscribe - 無限文字起こしを実行するか

* @param {boolean} sendEmail - メールを送信するか

* @return {Object} 文字起こし結果とセグメントファイルID

*/

function transcribeAudioFileWithSplit(fileId, splitMinutes, prompt, mode, infiniteTranscribe) {

try {

// リトライ付きでファイルを取得

const file = retryDriveOperation(function() {

return DriveApp.getFileById(fileId);

});

if (!file) {

return {

success: false,

error: 'ファイルが見つかりませんでした'

};

}

  

// 音声ファイルをBlobとして取得（リトライ付き）

const audioBlob = retryDriveOperation(function() {

return file.getBlob();

});

Logger.log('音声ファイル取得完了: ' + file.getName() + ' (サイズ: ' + audioBlob.getBytes().length + ' bytes)');

// モードのデフォルト値を設定

mode = mode || 'normal';

infiniteTranscribe = infiniteTranscribe !== undefined ? infiniteTranscribe : false;

let transcript;

let segments = [];

// 高速モードの場合：5分ごとに分割して文字起こし

if (mode === 'fast') {

Logger.log('高速モード：5分ごとに分割して文字起こしを開始');

const fastSplitMinutes = 5;

transcript = transcribeAudioFileFastMode(audioBlob, file.getName(), prompt || '', fastSplitMinutes);

if (!transcript || transcript.error) {

return {

success: false,

error: transcript && transcript.error ? transcript.error : '文字起こしに失敗しました'

};

}

segments = transcript.segments || [];

transcript = transcript.fullText || '';

} else {

// 通常モード：全体を一度に文字起こし

try {

transcript = transcribeWithGroq(audioBlob, file.getName(), prompt || '');

} catch (error) {

Logger.log('文字起こしエラー: ' + error.toString());

return {

success: false,

error: '文字起こしに失敗しました: ' + error.toString()

};

}

if (!transcript) {

return {

success: false,

error: '文字起こしに失敗しました。Groq APIの設定を確認してください。'

};

}

}

  

// 元のファイルと同じフォルダを取得（リトライ付き）

const parentFolders = retryDriveOperation(function() {

return file.getParents();

});

let parentFolder = null;

if (parentFolders.hasNext()) {

parentFolder = retryDriveOperation(function() {

return parentFolders.next();

});

}

  

// 日付と時間を含むフォルダを作成

const now = new Date();

const timeZone = Session.getScriptTimeZone();

const dateStr = Utilities.formatDate(now, timeZone, 'yyyy-MM-dd');

const timeStr = Utilities.formatDate(now, timeZone, 'HH-mm-ss');

const folderName = dateStr + '_' + timeStr + '_文字起こし';

let dateTimeFolder;

try {

// 既存のフォルダをチェック（同じ名前のフォルダが存在する場合）

const existingFolders = parentFolder ? parentFolder.getFoldersByName(folderName) : DriveApp.getFoldersByName(folderName);

if (existingFolders.hasNext()) {

dateTimeFolder = existingFolders.next();

Logger.log('既存の日時フォルダを使用: ' + folderName);

} else {

// 新しいフォルダを作成

dateTimeFolder = retryDriveOperation(function() {

if (parentFolder) {

return parentFolder.createFolder(folderName);

} else {

return DriveApp.createFolder(folderName);

}

});

Logger.log('新しい日時フォルダを作成: ' + folderName);

}

} catch (error) {

Logger.log('フォルダ作成エラー、親フォルダに保存: ' + error);

dateTimeFolder = parentFolder; // エラー時は親フォルダを使用

}

  

// 通常モードの場合のみ、文字起こし結果を時間ごとに分割

if (mode === 'normal' && segments.length === 0) {

// 1分あたりの文字数を推定（日本語の場合、1分で約300文字程度）

const charsPerMinute = 300;

const charsPerSegment = charsPerMinute * splitMinutes;

const transcriptLength = transcript.length;

let segmentIndex = 1;

// セグメントファイルを作成

for (let i = 0; i < transcriptLength; i += charsPerSegment) {

const segmentText = transcript.substring(i, Math.min(i + charsPerSegment, transcriptLength));

// セグメント番号を3桁にゼロパディング

const segmentNumber = ('000' + segmentIndex).slice(-3);

const segmentFileName = file.getName().replace(/\.[^/.]+$/, '') +

'_文字起こし_セグメント' + segmentNumber + '.txt';

// セグメントファイルを作成（リトライ付き）

const segmentFile = retryDriveOperation(function() {

return DriveApp.createFile(segmentFileName, segmentText, MimeType.PLAIN_TEXT);

});

if (dateTimeFolder) {

retryDriveOperation(function() {

segmentFile.moveTo(dateTimeFolder);

}, 3, 500);

} else if (parentFolder) {

retryDriveOperation(function() {

segmentFile.moveTo(parentFolder);

}, 3, 500);

}

// ファイル作成間に少し待機

Utilities.sleep(100);

segments.push({

index: segmentIndex,

fileName: segmentFileName,

fileId: segmentFile.getId(),

text: segmentText,

startChar: i,

endChar: Math.min(i + charsPerSegment, transcriptLength)

});

segmentIndex++;

}

}

  

// カスタムプロンプトで処理（プロンプトが指定されている場合）

let processedTranscript = transcript;

if (prompt && prompt.trim() !== '') {

try {

Logger.log('カスタムプロンプトで処理開始');

processedTranscript = processWithGroqChat(transcript, prompt.trim());

Logger.log('カスタムプロンプト処理完了');

} catch (error) {

Logger.log('カスタムプロンプト処理エラー: ' + error);

// エラー時は元の文字起こし結果を使用

processedTranscript = transcript;

}

}

// 無限文字起こし（AI処理）を実行（常に有効）

let finalTranscript = processedTranscript;

let htmlTranscript = '';

try {

Logger.log('無限文字起こし（AI処理）を開始');

const infiniteResult = processInfiniteTranscribe(processedTranscript);

finalTranscript = infiniteResult.text;

htmlTranscript = infiniteResult.html;

Logger.log('無限文字起こし完了');

} catch (error) {

Logger.log('無限文字起こしエラー: ' + error);

// エラー時は元のテキストを使用し、マークダウンをHTMLに変換

htmlTranscript = markdownToHtml(processedTranscript);

}

// テキストファイル用に読みやすい形式に変換

const formattedText = formatTextForFile(finalTranscript);

  

// 全体のテキストファイルも作成（リトライ付き）

const fileName = file.getName().replace(/\.[^/.]+$/, '') + '_文字起こし_全体.txt';

const textFile = retryDriveOperation(function() {

return DriveApp.createFile(fileName, formattedText, MimeType.PLAIN_TEXT);

});

// HTMLファイルも作成

const htmlFileName = file.getName().replace(/\.[^/.]+$/, '') + '_文字起こし_全体.html';

const htmlContent = createHtmlDocument(htmlTranscript, file.getName());

const htmlFile = retryDriveOperation(function() {

return DriveApp.createFile(htmlFileName, htmlContent, MimeType.HTML);

});

if (dateTimeFolder) {

retryDriveOperation(function() {

textFile.moveTo(dateTimeFolder);

htmlFile.moveTo(dateTimeFolder);

}, 3, 500);

} else if (parentFolder) {

retryDriveOperation(function() {

textFile.moveTo(parentFolder);

htmlFile.moveTo(parentFolder);

}, 3, 500);

}

  

return {

success: true,

transcript: finalTranscript,

htmlTranscript: htmlTranscript,

originalTranscript: transcript,

textFileId: textFile.getId(),

htmlFileId: htmlFile.getId(),

textFileName: fileName,

htmlFileName: htmlFileName,

folderId: dateTimeFolder ? dateTimeFolder.getId() : (parentFolder ? parentFolder.getId() : null),

folderName: folderName,

segmentFiles: segments,

segmentCount: segments.length

};

} catch (error) {

Logger.log('transcribeAudioFileWithSplit エラー: ' + error);

const errorMessage = error.toString();

// レート制限エラーの場合、より分かりやすいメッセージを返す

if (errorMessage.includes('混雑') || errorMessage.includes('集中')) {

return {

success: false,

error: 'Google Driveへのアクセスが集中しています。しばらく待ってから再度お試しください。'

};

}

return {

success: false,

error: errorMessage

};

}

}

  

/**

* フォルダ内の文字起こしファイルを結合

* @param {string} folderId - Google DriveフォルダID

* @return {Object} 結合結果

*/

function combineTranscriptFiles(folderId) {

try {

// リトライ付きでフォルダを取得

const folder = retryDriveOperation(function() {

return DriveApp.getFolderById(folderId);

});

if (!folder) {

return {

success: false,

error: 'フォルダが見つかりませんでした'

};

}

  

// セグメントファイルを検索（_文字起こし_セグメントで始まるファイル）

const files = retryDriveOperation(function() {

return folder.getFiles();

});

const segmentFiles = [];

const baseNames = {}; // 元のファイル名ごとにグループ化

  

while (files.hasNext()) {

// リトライ付きでファイルを取得

const file = retryDriveOperation(function() {

return files.next();

}, 3, 500);

const fileName = file.getName();

// セグメントファイルかチェック

if (fileName.includes('_文字起こし_セグメント')) {

// 元のファイル名を抽出

const baseNameMatch = fileName.match(/^(.+?)_文字起こし_セグメント/);

if (baseNameMatch) {

const baseName = baseNameMatch[1];

if (!baseNames[baseName]) {

baseNames[baseName] = [];

}

// セグメント番号を抽出

const segmentMatch = fileName.match(/セグメント(\d+)/);

const segmentIndex = segmentMatch ? parseInt(segmentMatch[1]) : 0;

baseNames[baseName].push({

index: segmentIndex,

file: file,

fileName: fileName

});

}

}

}

  

let combinedCount = 0;

const combinedFileIds = [];

  

// 各ベース名ごとに結合

for (const baseName in baseNames) {

const segments = baseNames[baseName];

// セグメント番号でソート

segments.sort(function(a, b) {

return a.index - b.index;

});

  

// 各セグメントの内容を読み込んで結合（リトライ付き）

let combinedText = '';

segments.forEach(function(segment) {

const content = retryDriveOperation(function() {

return segment.file.getBlob().getDataAsString();

}, 3, 500);

combinedText += content + '\n\n';

// ファイル読み込み間に少し待機

Utilities.sleep(50);

});

  

// 結合ファイルを作成（リトライ付き）

const combinedFileName = baseName + '_文字起こし_結合.txt';

const combinedFile = retryDriveOperation(function() {

return DriveApp.createFile(combinedFileName, combinedText.trim(), MimeType.PLAIN_TEXT);

});

retryDriveOperation(function() {

/**

* @param {string} fileName - ファイル名

* @param {string} prompt - カスタムプロンプト

* @param {number} splitMinutes - 分割時間（分、デフォルト5分）

* @return {Object} 文字起こし結果とセグメント情報

*/

function transcribeAudioFileFastMode(audioBlob, fileName, prompt, splitMinutes) {

splitMinutes = splitMinutes || 5;

Logger.log('高速モード開始: ' + splitMinutes + '分ごとに分割');

try {

// 音声ファイルの長さを推定（正確な長さは取得できないため、ファイルサイズから推定）

// MP3の場合、約1MB = 1分程度と仮定

const fileSizeMB = audioBlob.getBytes().length / (1024 * 1024);

const estimatedMinutes = Math.max(1, Math.ceil(fileSizeMB));

const segmentCount = Math.ceil(estimatedMinutes / splitMinutes);

Logger.log('推定ファイル長: ' + estimatedMinutes + '分、セグメント数: ' + segmentCount);

// 実際には、Groq APIは音声ファイル全体を処理するため、

// ここでは全体を文字起こししてから、結果を5分ごとに分割する

const fullTranscript = transcribeWithGroq(audioBlob, fileName, prompt || '');

if (!fullTranscript) {

return { error: '文字起こしに失敗しました' };

}

// 文字起こし結果を5分ごとに分割（1分あたり約300文字と仮定）

// 実際の高速化のため、結果を即座に返す

const charsPerMinute = 300;

const charsPerSegment = charsPerMinute * splitMinutes;

const segments = [];

const transcriptLength = fullTranscript.length;

let segmentIndex = 1;

// 結果を5分ごとに分割

for (let i = 0; i < transcriptLength; i += charsPerSegment) {

const segmentText = fullTranscript.substring(i, Math.min(i + charsPerSegment, transcriptLength));

segments.push({

index: segmentIndex,

text: segmentText,

startChar: i,

endChar: Math.min(i + charsPerSegment, transcriptLength)

});

segmentIndex++;

}

Logger.log('高速モード完了: ' + segments.length + '個のセグメントを作成（合計文字数: ' + transcriptLength + '）');

Logger.log('高速モード完了: ' + segments.length + '個のセグメントを作成');

};

} catch (error) {

Logger.log('高速モードエラー: ' + error);

return { error: error.toString() };

}

}

  

/**

const parts = [];

parts.push({

if (prompt && prompt.trim() !== '') {

parts.push({

name: 'prompt',

value: prompt.trim()

});

Logger.log('カスタムプロンプトを使用: ' + prompt.substring(0, 100) + '...');

}

// file パラメータ

parts.push({

name: 'file',

blob: audioBlob,

});

// multipart/form-data形式のペイロードを作成

const payload = createMultipartPayload(parts, boundary);

// APIリクエストを送信

Logger.log('Groq API リクエスト送信開始');

Logger.log('URL: ' + apiUrl);

Logger.log('ファイル名: ' + fileName);

Logger.log('ファイルサイズ: ' + audioBlob.getBytes().length + ' bytes');

Logger.log('Boundary: ' + boundary);

// multipart/form-dataリクエストを送信

// Content-Typeヘッダーにboundaryを指定する必要がある

const response = UrlFetchApp.fetch(apiUrl, {

const responseCode = response.getResponseCode();

const responseText = response.getContentText();

  

Logger.log('Groq API レスポンスコード: ' + responseCode);

/**

* multipart/form-data形式のペイロードを作成

* @param {Array} parts - フォームパーツの配列

* @param {string} boundary - バウンダリ文字列

* @return {Blob} multipart/form-data形式のBlob

*/

function createMultipartPayload(parts, boundary) {

const delimiter = '\r\n--' + boundary + '\r\n';

const closeDelimiter = '\r\n--' + boundary + '--\r\n';

let payloadParts = [];

parts.forEach(function(part) {

let partData = '';

if (part.blob) {

// ファイルパート

partData += 'Content-Disposition: form-data; name="' + part.name + '"; filename="' + part.filename + '"\r\n';

partData += 'Content-Type: ' + part.blob.getContentType() + '\r\n\r\n';

// テキスト部分とバイナリ部分を結合

const textBytes = Utilities.newBlob(partData).getBytes();

const blobBytes = part.blob.getBytes();

payloadParts.push({

type: 'bytes',

data: textBytes.concat(blobBytes)

});

} else {

// 通常のパラメータ

partData += 'Content-Disposition: form-data; name="' + part.name + '"\r\n\r\n';

partData += part.value;

payloadParts.push({

type: 'string',

data: partData

});

}

});

// すべてのパーツを結合

let allBytes = [];

// 最初のバウンダリ（--で始まる）

const firstBoundaryBytes = Utilities.newBlob('--' + boundary).getBytes();

allBytes = allBytes.concat(firstBoundaryBytes);

payloadParts.forEach(function(part, index) {

if (index > 0) {

const delimiterBytes = Utilities.newBlob(delimiter).getBytes();

allBytes = allBytes.concat(delimiterBytes);

} else {

// 最初のパートの前にも改行を追加

const firstDelimiterBytes = Utilities.newBlob('\r\n').getBytes();

allBytes = allBytes.concat(firstDelimiterBytes);

}

if (part.type === 'bytes') {

allBytes = allBytes.concat(part.data);

} else {

const stringBytes = Utilities.newBlob(part.data).getBytes();

allBytes = allBytes.concat(stringBytes);

}

});

// クロージングデリミタを追加

const closeBytes = Utilities.newBlob(closeDelimiter).getBytes();

allBytes = allBytes.concat(closeBytes);

return Utilities.newBlob(allBytes);

}

/**

* テキストファイル用に読みやすい形式に変換

* @param {string} text - 元のテキスト

* @return {string} フォーマット済みテキスト

*/

function formatTextForFile(text) {

if (!text) return '';

let formatted = text;

// マークダウンのテーブル形式（||で区切られた形式）を読みやすい形式に変換

// 例: || 内容 || → 【内容】

formatted = formatted.replace(/\|\|\s*([^|]+?)\s*\|\|/g, '【$1】');

// 見出しを明確に（レベルに応じて）

formatted = formatted.replace(/^####\s+(.+)$/gm, '\n【$1】\n');

formatted = formatted.replace(/^###\s+(.+)$/gm, '\n■ $1\n');

formatted = formatted.replace(/^##\s+(.+)$/gm, '\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n■ $1\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n');

formatted = formatted.replace(/^#\s+(.+)$/gm, '\n═══════════════════════════════════════════════════════════\n$1\n═══════════════════════════════════════════════════════════\n');

// 太字を明確に

formatted = formatted.replace(/\*\*(.+?)\*\*/g, '【$1】');

// 箇条書きを明確に

formatted = formatted.replace(/^[-*]\s+(.+)$/gm, ' • $1');

// 番号付きリスト

formatted = formatted.replace(/^\d+\.\s+(.+)$/gm, ' $&');

// 段落の区切りを明確にする（空行を1行に統一）

formatted = formatted.replace(/\n{3,}/g, '\n\n');

// 各行の前後の空白を整理

const lines = formatted.split('\n');

const cleanedLines = [];

for (let i = 0; i < lines.length; i++) {

const line = lines[i].trim();

if (line || (i > 0 && cleanedLines[cleanedLines.length - 1] !== '')) {

cleanedLines.push(line);

}

}

formatted = cleanedLines.join('\n');

return formatted.trim();

}

  

/**

* マークダウン記号をHTMLに変換

* @param {string} text - マークダウンテキスト

* @return {string} HTMLテキスト

*/

function markdownToHtml(text) {

if (!text) return '';

const lines = text.split('\n');

let result = [];

let inList = false;

let inParagraph = false;

for (let i = 0; i < lines.length; i++) {

const line = lines[i];

const trimmedLine = line.trim();

// 空行

if (!trimmedLine) {

if (inList) {

result.push('</ul>');

inList = false;

}

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

continue;

}

// 見出し（#### → h4, ### → h3, ## → h2, # → h1）

if (trimmedLine.match(/^####\s+(.+)$/)) {

if (inList) {

result.push('</ul>');

inList = false;

}

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

result.push('<h4>' + trimmedLine.replace(/^####\s+/, '') + '</h4>');

continue;

}

if (trimmedLine.match(/^###\s+(.+)$/)) {

if (inList) {

result.push('</ul>');

inList = false;

}

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

result.push('<h3>' + trimmedLine.replace(/^###\s+/, '') + '</h3>');

continue;

}

if (trimmedLine.match(/^##\s+(.+)$/)) {

if (inList) {

result.push('</ul>');

inList = false;

}

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

result.push('<h2>' + trimmedLine.replace(/^##\s+/, '') + '</h2>');

continue;

}

if (trimmedLine.match(/^#\s+(.+)$/)) {

if (inList) {

result.push('</ul>');

inList = false;

}

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

result.push('<h1>' + trimmedLine.replace(/^#\s+/, '') + '</h1>');

continue;

}

// 箇条書き（- または * で始まる行）

if (trimmedLine.match(/^[-*]\s+(.+)$/)) {

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

if (!inList) {

result.push('<ul>');

inList = true;

}

const itemText = trimmedLine.replace(/^[-*]\s+/, '');

// 太字と斜体を処理

let processedItem = itemText.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');

processedItem = processedItem.replace(/(?<!\*)\*([^*]+?)\*(?!\*)/g, '<em>$1</em>');

result.push('<li>' + processedItem + '</li>');

continue;

}

// 番号付きリスト

if (trimmedLine.match(/^\d+\.\s+(.+)$/)) {

if (inParagraph) {

result.push('</p>');

inParagraph = false;

}

if (!inList) {

result.push('<ol>');

inList = true;

}

const itemText = trimmedLine.replace(/^\d+\.\s+/, '');

// 太字と斜体を処理

let processedItem = itemText.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');

processedItem = processedItem.replace(/(?<!\*)\*([^*]+?)\*(?!\*)/g, '<em>$1</em>');

result.push('<li>' + processedItem + '</li>');

continue;

}

// 通常のテキスト行

if (inList) {

result.push('</ul>');

inList = false;

}

if (!inParagraph) {

result.push('<p>');

inParagraph = true;

}

// 太字と斜体を処理

let processedLine = trimmedLine.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>');

processedLine = processedLine.replace(/(?<!\*)\*([^*]+?)\*(?!\*)/g, '<em>$1</em>');

result.push(processedLine);

}

// 閉じタグ

if (inList) {

result.push('</ul>');

}

if (inParagraph) {

result.push('</p>');

}

return result.join('\n');

}
